# IMPORTANT: picamera2 and libcamera are NOT pip-installable on a Raspberry Pi.
# They are built against the system libcamera stack and ship via apt. The venv
# must therefore be able to see system site-packages:
#
#   sudo apt install -y python3-picamera2 python3-libcamera python3-numpy python3-flask python3-serial
#   python3 -m venv --system-site-packages .venv
#
# An existing venv can be opened up by editing its pyvenv.cfg:
#   include-system-site-packages = true
#
# Then install the pip-only dependency below:
#   .venv/bin/pip install -r requirements.txt
